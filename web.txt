        });

        // Tambahkan efek hover ke semua elemen interaktif
        const interactives = document.querySelectorAll('a, button, .bento-card, .project-card, .gallery-item, .team-card, input, textarea, .tag');
        interactives.forEach(el => {
            el.addEventListener('mouseenter', () => document.body.classList.add('hovering'));
            el.addEventListener('mouseleave', () => document.body.classList.remove('hovering'));
        });

        function handleNavClick(e, name, desc) {
            e.preventDefault();
            alert("Menu: " + name + "\n\n" + desc);
            const target = e.currentTarget.getAttribute('href');
            setTimeout(() => {
                document.querySelector(target).scrollIntoView({behavior: 'smooth'});
                document.querySelector('.nav-links').classList.remove('active');
            }, 100);
        }
