        // --- FUNGSI NAVIGASI (8 TOMBOL) ---
        function handleNavClick(event, sectionName, description) {
            event.preventDefault(); // Mencegah langsung scroll dulu
            
            // Tampilkan alert interaktif
            alert("Anda menekan menu: " + sectionName + "\n\nKeterangan: " + description + "\n\nMenggulir ke halaman tersebut...");
            
            // Ambil ID tujuan dari href
            const targetId = event.currentTarget.getAttribute('href');
            
            // Scroll manual setelah OK ditekan
            setTimeout(() => {
                document.querySelector(targetId).scrollIntoView({
                    behavior: 'smooth'
                });
                
                // Tutup menu mobile jika terbuka
                document.querySelector('.nav-links').classList.remove('active');
            }, 100);
        }

        // --- FUNGSI PROJECT DETAILS ---
        function showProjectDetails(title, details) {
            alert("DETAIL PROJECT: " + title.toUpperCase() + "\n\n" + details);
        }

            event.target.reset(); // Kosongkan form
        }

        // Toggle Mobile Menu
        function toggleMenu() {
            const navLinks = document.querySelector('.nav-links');
            navLinks.classList.toggle('active');
        }

        // Close menu when clicking a link (untuk mobile)
        document.querySelectorAll('.nav-links a').forEach(link => {

                    document.querySelector('.nav-links').classList.remove('active');
                }
            });
        });
